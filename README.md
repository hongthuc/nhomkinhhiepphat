        // 3. Hệ thống tự động dự phòng (fallback) cho ảnh Google Drive để luôn hiển thị ổn định
        function attachImageFallbacks() {
            const imgs = document.querySelectorAll('img.gdrive-img');
            imgs.forEach(function (img) {
                if (img.dataset.fallbackAttached) return;
                img.dataset.fallbackAttached = 'true';
                let step = 0;
                img.addEventListener('error', function () {
                    const id = img.dataset.gdriveId;
                    const label = img.dataset.fallbackLabel || 'Nhom+Kinh+Hiep+Phat';
                    step++;
                    if (step === 1 && id) {
                        img.src = 'https://lh3.googleusercontent.com/d/' + id + '=w1000';
                    } else if (step === 2 && id) {
                        img.src = 'https://drive.google.com/uc?export=view&id=' + id;
                    } else if (step === 3) {
                        img.src = 'https://placehold.co/600x450/0D3B66/FFFFFF?text=' + label;
                    }
                });
            });
        }

        function selectQuoteCategory(label) {
            const selectEl = document.getElementById('qCategory');
            const noteEl = document.getElementById('qNote');
            if (!selectEl) return;
            let matched = false;
            for (let i = 0; i < selectEl.options.length; i++) {
                if (label.toLowerCase().includes(selectEl.options[i].value.toLowerCase())) {
                    selectEl.selectedIndex = i;
                    matched = true;
                    break;
                }
            }
            if (!matched && noteEl) {
                noteEl.value = 'Quan tâm sản phẩm: ' + label;
            }
        }

        // 9. Xử lý Banner 2 slide tự trượt + vuốt cảm ứng trên điện thoại
        let currentSlide = 0;
        const slides = document.querySelectorAll('.hero-slide-item');
        const dots = document.querySelectorAll('.hero-dot');
        let slideTimer = null;

        function showSlide(index) {
            currentSlide = (index + slides.length) % slides.length;
            slides.forEach(function (slide, idx) {
                if (idx === currentSlide) {
                    slide.classList.add('active');
                } else {
                    slide.classList.remove('active');
                }
            });
            dots.forEach(function (dot, idx) {
                if (idx === currentSlide) {
                    dot.classList.add('w-6', 'bg-brand-gold');
                    dot.classList.remove('w-2.5', 'bg-white/50');
                } else {
                    dot.classList.remove('w-6', 'bg-brand-gold');
                    dot.classList.add('w-2.5', 'bg-white/50');
                }
            });
        }

        function resetAutoSlide() {
            if (slideTimer) clearInterval(slideTimer);
            slideTimer = setInterval(function () {
                showSlide(currentSlide + 1);
            }, 5500);
        }

        document.getElementById('prevSlideBtn').addEventListener('click', function () {
            showSlide(currentSlide - 1);
            resetAutoSlide();
        });
        document.getElementById('nextSlideBtn').addEventListener('click', function () {
            showSlide(currentSlide + 1);
            resetAutoSlide();
        });
        dots.forEach(function (dot) {
            dot.addEventListener('click', function () {
                showSlide(parseInt(dot.dataset.slide, 10));
                resetAutoSlide();
            });
        });

        // Hỗ trợ vuốt ngang trên điện thoại cho Banner
        const sliderEl = document.getElementById('heroSlider');
        let touchStartX = 0;
        sliderEl.addEventListener('touchstart', function (e) {
            touchStartX = e.changedTouches[0].screenX;
        }, { passive: true });
        sliderEl.addEventListener('touchend', function (e) {
            const diff = e.changedTouches[0].screenX - touchStartX;
            if (Math.abs(diff) > 45) {
                showSlide(diff < 0 ? currentSlide + 1 : currentSlide - 1);
                resetAutoSlide();
            }
        }, { passive: true });
